cmake_minimum_required(VERSION 3.20)
project(Kadrolov VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(KADROLOV_BUILD_APP "Build desktop app" ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Test PrintSupport)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Test PrintSupport)
if(KADROLOV_BUILD_APP)
  add_executable(kadrolov MACOSX_BUNDLE WIN32
    desktop/src/main.cpp desktop/src/MainWindow.cpp desktop/src/MainWindow.h
    desktop/resources.qrc
    desktop/src/Editor.cpp desktop/src/Editor.h
    desktop/src/RegionSelector.cpp desktop/src/RegionSelector.h
    desktop/src/RecordingPanel.cpp desktop/src/RecordingPanel.h
    desktop/src/Recorder.cpp desktop/src/Recorder.h
    desktop/src/GlobalHotkeys.cpp desktop/src/GlobalHotkeys.h
    desktop/src/Autostart.cpp desktop/src/Autostart.h
    desktop/src/DeviceInventory.cpp desktop/src/DeviceInventory.h
    desktop/src/Recovery.cpp desktop/src/Recovery.h
    desktop/src/ClickOverlay.cpp desktop/src/ClickOverlay.h
    desktop/src/RecordingPlan.cpp desktop/src/RecordingPlan.h desktop/src/CaptureOptions.h)
  target_include_directories(kadrolov PRIVATE desktop/src)
  target_compile_definitions(kadrolov PRIVATE KADROLOV_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  target_link_libraries(kadrolov PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::PrintSupport)
  set_target_properties(kadrolov PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Кадролов" MACOSX_BUNDLE_GUI_IDENTIFIER "ru.kadrolov.desktop" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/native/macos/Info.plist.in")
  if(APPLE)
    target_sources(kadrolov PRIVATE assets/icons/kadrolov.icns)
    set_source_files_properties(assets/icons/kadrolov.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_link_libraries(kadrolov PRIVATE "-framework CoreGraphics" "-framework Carbon")
    add_custom_target(mac_capture ALL COMMAND bash "${CMAKE_SOURCE_DIR}/native/macos/build.sh" "${CMAKE_BINARY_DIR}/native/kadrolov-capture" BYPRODUCTS "${CMAKE_BINARY_DIR}/native/kadrolov-capture")
    add_dependencies(kadrolov mac_capture)
    add_custom_command(TARGET kadrolov POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_BINARY_DIR}/native/kadrolov-capture" "$<TARGET_FILE_DIR:kadrolov>/kadrolov-capture")
    add_executable(recorder_smoke tests/recorder_smoke.cpp desktop/src/Recorder.cpp desktop/src/Recorder.h desktop/src/RecordingPlan.cpp desktop/src/ClickOverlay.cpp desktop/src/ClickOverlay.h)
    target_include_directories(recorder_smoke PRIVATE desktop/src)
    target_link_libraries(recorder_smoke PRIVATE Qt${QT_VERSION_MAJOR}::Widgets "-framework CoreGraphics")
    target_compile_definitions(recorder_smoke PRIVATE KADROLOV_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_dependencies(recorder_smoke mac_capture)
  endif()
  if(UNIX AND NOT APPLE)
    find_package(X11 QUIET)
    if(X11_FOUND)
      target_compile_definitions(kadrolov PRIVATE KADROLOV_HAVE_X11=1)
      target_include_directories(kadrolov PRIVATE ${X11_INCLUDE_DIR})
      target_link_libraries(kadrolov PRIVATE ${X11_LIBRARIES})
    endif()
    add_custom_command(TARGET kadrolov POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/native/linux/portal_capture.py" "$<TARGET_FILE_DIR:kadrolov>/portal_capture.py")
    install(PROGRAMS native/linux/portal_capture.py DESTINATION bin)
    install(FILES packaging/linux/kadrolov.desktop DESTINATION share/applications)
    install(FILES assets/icons/kadrolov-256.png DESTINATION share/icons/hicolor/256x256/apps RENAME kadrolov.png)
    install(FILES assets/icons/kadrolov.svg DESTINATION share/icons/hicolor/scalable/apps)
  endif()
  if(WIN32)
    configure_file(native/windows/app.rc.in ${CMAKE_BINARY_DIR}/kadrolov.rc @ONLY)
    target_sources(kadrolov PRIVATE ${CMAKE_BINARY_DIR}/kadrolov.rc)
    add_executable(kadrolov-audio native/windows/audio.cpp)
    target_link_libraries(kadrolov-audio PRIVATE ole32 uuid)
    target_compile_definitions(kadrolov-audio PRIVATE UNICODE _UNICODE)
    add_executable(kadrolov-capture native/windows/capture.cpp)
    target_link_libraries(kadrolov-capture PRIVATE mf mfplat mfreadwrite mfuuid ole32 uuid gdi32 user32)
    target_compile_definitions(kadrolov-capture PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(kadrolov PRIVATE user32)
  endif()
  install(TARGETS kadrolov BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION share/doc/kadrolov)
  install(DIRECTORY licenses/ DESTINATION share/doc/kadrolov/licenses)
endif()
enable_testing()
add_executable(recording_plan_test tests/recording_plan_test.cpp desktop/src/RecordingPlan.cpp)
target_include_directories(recording_plan_test PRIVATE desktop/src)
target_link_libraries(recording_plan_test PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
add_test(NAME recording_plan COMMAND recording_plan_test)
if(EXISTS "${CMAKE_SOURCE_DIR}/desktop/tests/editor_test.cpp" AND KADROLOV_BUILD_APP)
  add_executable(editor_test desktop/tests/editor_test.cpp desktop/src/Editor.cpp desktop/src/Editor.h)
  target_include_directories(editor_test PRIVATE desktop/src)
  target_link_libraries(editor_test PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test Qt${QT_VERSION_MAJOR}::PrintSupport)
  add_test(NAME editor COMMAND editor_test)
endif()
